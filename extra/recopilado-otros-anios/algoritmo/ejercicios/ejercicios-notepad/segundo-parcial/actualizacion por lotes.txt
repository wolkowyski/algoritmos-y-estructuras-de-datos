	Mientras (clave_mae<> HV) o (clave_mov <> HV) hacer
		Si (clave_mae < clave_mov ) entonces
			reg_sal:=reg_mae
			esc (sal,reg_Sal)

		Fin si
	Fin mientras

Subaccion PROCESOS IGUALES ES :
AUX := reg_mae
resg_clave:=reg_mov.clave_mov
	Mientras (resg_clave = reg_mov.clave_mov) Hacer
		Segun (reg_mov.cod_mov) hacer
			1: esc ('error alta')
			2: modificar
			3: baja
		Fin segun
		Leer_mov
	Fin Mientras
	esc (sal,aux)
	Leer_mae
Fin sub
