Subaccion guardar (prim: puntero a nodo; origen: N(5))es
	nuevo (q)
	^q.dato:=origen
	prim:=q;
	Si (p = nil) entonces
		^q.prox:=nil
		prim:=q
		SINO
			p:= prim
			Mientas (p<> nil)hacer
				p:=^p.prox
			Fin Mientas
			Si (p=nil) entonces
				^p.prox:= q;
				^q.prox:= nil
			Fin si
	Fin si
Fin subaccion
